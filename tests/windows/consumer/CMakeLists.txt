cmake_minimum_required(VERSION 3.18)
project(WalletCoreWindowsConsumer C CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 20)
find_package(WalletCore CONFIG REQUIRED)

foreach(kind IN ITEMS Shared Static)
    foreach(language IN ITEMS c cpp)
        add_executable(consumer_${kind}_${language} main.c)
        if(language STREQUAL cpp)
            # Compile the same public C API consumer as C++ in a separate target.
            configure_file(main.c ${CMAKE_CURRENT_BINARY_DIR}/main.cpp COPYONLY)
            set_property(TARGET consumer_${kind}_${language} PROPERTY SOURCES ${CMAKE_CURRENT_BINARY_DIR}/main.cpp)
        endif()
        target_link_libraries(consumer_${kind}_${language} PRIVATE WalletCore::${kind})
        target_compile_options(consumer_${kind}_${language} PRIVATE /utf-8)
    endforeach()
endforeach()
add_executable(consumer_dynamic main.c)
target_compile_definitions(consumer_dynamic PRIVATE TW_DYNAMIC_LOAD NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(consumer_dynamic PRIVATE ${WALLET_CORE_PACKAGE}/include)
